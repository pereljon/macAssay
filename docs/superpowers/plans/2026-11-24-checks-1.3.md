# Checks 1.3 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build macassay 1.3 as designed in `dev/features/checks-1.3.md`: 14 themed report sections, stable finding IDs, hex masking, the approved severity changes, and about 20 groups of new checks.

**Architecture:** `macassay` is one Bash 3.2 script. Pure parsers live in the "logic functions" block and are tested by `dev/test.sh`, which sources the script with `MACASSAY_LIB=1`. The audit body writes evidence to `$BODY` and findings to `$FIND`, then assembles the report. Commit 1 only moves code; commit 2 only renames; later commits change behavior, one section at a time.

**Tech Stack:** `/bin/bash` 3.2, stock macOS tools (`awk`, `sed`, `plutil`, `defaults`, `dscl`, `stat`, `find`, `lsof`, `security`, `codesign`), ShellCheck for linting (development only).

## Global Constraints

- Run under `/bin/bash` 3.2; call only tools that ship with macOS. No `mapfile`, no associative arrays, no `${var,,}`.
- No new sudo rules and no new side effects. `gen_rules` and the `priv` calls stay unchanged.
- No secret value reaches the report, the temporary folder, or the terminal. Read credential files only through `has_key` and `stat -L`. Never run `ssh-keygen` on a private key. Never write `ps` command lines to `$TMP`.
- Every finding call is `finding ID SEVERITY LABEL TEXT` with a literal ID `MA-<LABEL>-NNN`, or `"$id"` read from a function that prints literal IDs.
- One item gives one finding, at its highest severity. Repeated findings collapse into one finding with a count or a list.
- Use the project terminology: a check is a test, a finding is a judged result, a flag is a command-line option.
- Do not commit generated reports. Keep them in `.scratch/checks-1.3/`, which is gitignored.
- Do not write the user's name or initials in any project file.
- Commit on `feature/checks-1.3` only. Write each commit message to `.scratch/checks-1.3/msg.txt` first, ending with the attribution lines from the session, then run `git commit -F .scratch/checks-1.3/msg.txt` in its own Bash call.
- Before each commit, run `bash -n macassay`, `shellcheck macassay` (3 historical findings: SC2010 and SC2012; no new ones), and `dev/test.sh` (all pass).

# pam_modules FILE DEPTH: prints each module given as an absolute path outside /usr/lib/pam. Follows include and
# substack targets inside ${PAMD:-/etc/pam.d}; the PAM_SEEN list and a depth limit of 16 stop loops.
pam_modules() {
  local f="$1" depth="$2" ty ctl rest mod
  if [ "$depth" -gt 16 ] || [ ! -f "$f" ]; then return 0; fi
  printf '%s\n' "$PAM_SEEN" | grep -qxF -- "$f" && return 0
  PAM_SEEN="$PAM_SEEN"$'\n'"$f"
  while read -r ty rest; do
    case "$ty" in ''|'#'*) continue ;; esac
    if [ "${rest:0:1}" = "[" ]; then ctl="${rest%%]*}]"; rest="${rest#*]}"
    else ctl="${rest%%[[:space:]]*}"; rest="${rest#"$ctl"}"; fi

# forced_ext_ids FILE KEY: prints the IDs of force-installed extensions under KEY in a plist or JSON file.
# Chromium lists hold "ID;update URL" strings; Firefox ExtensionSettings maps an ID to its installation_mode.
forced_ext_ids() {
  plutil -extract "$2" xml1 -o - "$1" 2>/dev/null | plutil -p - 2>/dev/null | awk '
    /^  [0-9]+ => "/ { s = $0; sub(/^[^"]*"/, "", s); sub(/[;"].*/, "", s); print s; next }
    /^  "/ { k = $0; sub(/^  "/, "", k); sub(/" => .*/, "", k); next }
    /"installation_mode" => "force_installed"/ { print k }'; }
# forced_ext_finding ENROLLED; stdin: "source<TAB>browser<TAB>IDs" lines, source managed or local.
# Prints "ID<TAB>SEVERITY<TAB>browsers" for the one finding they give.
forced_ext_finding() {
  awk -F'\t' -v e="$1" '{ b[$2] = 1; if ($1 == "managed") m[$2] = 1 }
    END { for (k in b) n++; if (!n) exit
      if (e == 1) { id = "MA-APPS-007"; sv = "INFO"; for (k in b) l = l (l == "" ? "" : ",") k }
      else { for (k in m) l = l (l == "" ? "" : ",") k
        if (l != "") { id = "MA-APPS-005"; sv = "HIGH" } else { id = "MA-APPS-006"; sv = "MEDIUM"; for (k in b) l = l (l == "" ? "" : ",") k } }
      print id "\t" sv "\t" l }'; }

# has_key FILE PATTERN: succeeds when FILE has a line that matches PATTERN. Prints nothing; the only way credential files are read.
has_key() { grep -qsE -- "$2" "$1"; }
# cred_scan HOME: prints "path<TAB>mode owner<TAB>yes|no" for each credential file present; yes means an inline secret.
cred_scan() {
  local rel re f o s
  printf '%s\n' "$CRED_FILES" | while IFS=$'\t' read -r rel re; do
    f="$1/$rel"; [ -e "$f" ] || continue
    o=$(stat -L -f '%Sp %Su' "$f" 2>/dev/null) || continue
    if has_key "$f" "$re"; then s=yes; else s=no; fi
    printf '%s\t%s\t%s\n' "$rel" "$o" "$s"
  done; }
# ssh_key_state FILE: prints "no passphrase" or "passphrase" for a private key file, nothing for other files.
# Reads the header line and at most 31 characters of line 2; never decodes or runs a tool on the key.
ssh_key_state() {
  local l1
  l1=$(head -n 1 "$1" 2>/dev/null)
  case "$l1" in
    "-----BEGIN OPENSSH PRIVATE KEY-----")
      if [ "$(sed -n 2p "$1" | cut -c1-31)" = b3BlbnNzaC1rZXktdjEAAAAABG5vbmU ]; then echo "no passphrase"; else echo passphrase; fi ;;
    "-----BEGIN ENCRYPTED PRIVATE KEY-----") echo passphrase ;;
    "-----BEGIN "*"PRIVATE KEY-----") if head -n 3 "$1" | grep -q ENCRYPTED; then echo passphrase; else echo "no passphrase"; fi ;;
  esac; }
# ssh_pub_summary FILE: prints "file<TAB>bits<TAB>(TYPE)" for a public key, without its comment.
ssh_pub_summary() { ssh-keygen -l -f "$1" 2>/dev/null | awk -v f="${1##*/}" '{print f "\t" $1 "\t" $NF}'; }
# ssh_cfg FILE DEPTH SCOPE: prints "SCOPE ForwardAgent" for ForwardAgent yes, "SCOPE StrictHostKeyChecking" for
# StrictHostKeyChecking no or off, and "block" for each Host or Match block that is not for every host.
# SCOPE is "all" before the first Host or Match line and under Host *, and "host" elsewhere.
# Follows Include like ssh does; the SSH_SEEN list and a depth limit of 16 stop loops. Config text is never run.
ssh_cfg() {
  local f="$1" depth="$2" scope="$3" line key val pat g base
  local -a pats
  if [ "$depth" -gt 16 ] || [ ! -f "$f" ] || [ ! -r "$f" ]; then return 0; fi
  printf '%s\n' "$SSH_SEEN" | grep -qxF -- "$f" && return 0
  SSH_SEEN="$SSH_SEEN"$'\n'"$f"
  case "$f" in /etc/ssh/*) base=/etc/ssh ;; *) base="$THOME/.ssh" ;; esac
  while IFS= read -r line || [ -n "$line" ]; do
    line="${line#"${line%%[![:space:]]*}"}"
    case "$line" in ''|'#'*) continue ;; esac
    key=$(printf '%s' "${line%%[[:space:]=]*}" | tr '[:upper:]' '[:lower:]')
    val="${line#*[[:space:]=]}"; val="${val#"${val%%[![:space:]=]*}"}"; val="${val%"${val##*[![:space:]]}"}"
    case "$key" in
      host) if [ "$val" = "*" ]; then scope=all; else scope=host; echo block; fi ;;
      match) case "$(printf '%s' "$val" | tr '[:upper:]' '[:lower:]')" in all) scope=all ;; *) scope=host; echo block ;; esac ;;
      include)
        read -ra pats <<< "$val"
        for pat in "${pats[@]}"; do
          case "$pat" in *%*|*'$'*) continue ;; "~/"*) pat="$THOME/${pat#\~/}" ;; /*) ;; *) pat="$base/$pat" ;; esac
          while IFS= read -r g; do ssh_cfg "$g" $((depth + 1)) "$scope"; done < <(compgen -G "$pat")
        done ;;
      forwardagent) [ "$(printf '%s' "$val" | tr '[:upper:]' '[:lower:]')" = yes ] && echo "$scope ForwardAgent" ;;
      stricthostkeychecking) case "$(printf '%s' "$val" | tr '[:upper:]' '[:lower:]')" in no|off) echo "$scope StrictHostKeyChecking" ;; esac ;;
    esac
  done < "$f"; }
